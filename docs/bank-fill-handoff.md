   - ระวัง: `b.longer` แมปตามข้อความ ใช้กับตัวลวงทุกข้อที่ข้อความตรงกัน · ห้ามทำให้ตัวลวงสองตัวความหมายซ้ำกัน

## SQL ที่ใช้บ่อย
```sql
-- ภาพรวมทุก topic: จำนวนข้อ, ข้อยาก, คนเลือกเอก, คนซื้อ
select t.code, t.name, (select count(*) from prep.questions q where q.topic=t.code and q.status='approved') n,
  (select count(*) from prep.profiles p where p.major=t.code) chose
from prep.topics t order by t.section, t.sort;
-- หมวดย่อยของวิชา
select s.code, s.name, s.weight, count(q.id) filter (where q.status='approved') n
from prep.subtopics s left join prep.questions q on q.subtopic=s.code
where s.topic='B3-SOC' group by 1,2,3, s.sort order by s.sort;
```
